\begin{frame}{Electricity prices enter firms and household budgets}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\color{clews}\Large\bfseries Intermediate use}\par\medskip
      \begin{itemize}\setlength\itemsep{4pt}
        \item Electricity input shares come from the social accounting matrix.
        \item Direct and upstream exposure determine an industry-specific cost push.
        \item The cost increase is represented as a reduction in $\TFP_m$.
      \end{itemize}
      \smallskip
      {\color{sub}\scriptsize Reduced-form pending a structural energy input.}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{ogc}\Large\bfseries Final use}\par\medskip
      \begin{itemize}\setlength\itemsep{4pt}
        \item The household share enters through the energy-good consumption price.
        \item A subsistence component produces income-group incidence.
        \item Mechanically generated revenue is recycled or explicitly financed.
      \end{itemize}
      \smallskip
      {\color{sub}\scriptsize Electricity self-use is removed before combining the routes.}
    \end{column}
  \end{columns}
  \source{paper/sections/04-framework.tex, ``The energy channel''; docs/design/energy-price-transmission.md.}
\end{frame}

% 5 ---------------------------------------------------------------------------
\begin{frame}{LCOE is a price proxy, not a marginal price}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      {\color{claret}\Large\bfseries Candidate marginal signal}\par\medskip
      The commodity-balance dual has the preferred marginal interpretation, but the
      Philippine values are sparse and unstable for long-run transmission.
      \medskip

      \textbf{Hygiene tests}
      \begin{itemize}\setlength\itemsep{4pt}
        \item solver resolution and alternate-optimum stability
        \item decomposition against cost floors
        \item traceability to sourced inputs
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{clews}\Large\bfseries Deployed signal}\par\medskip
      The application uses the reform/base ratio of a curated electricity-cost index or
      reconstructed levelized cost.
      \medskip

      \begin{block}{Interpretation}
        A resulting fixed point would establish consistency under the selected proxy,
        not a competitive-equilibrium electricity price.
      \end{block}
    \end{column}
  \end{columns}
  \source{paper/sections/04-framework.tex, ``The levelized cost of electricity'' and ``Shadow-price hygiene''; paper/sections/06-calibration.tex.}
\end{frame}

% 6 ---------------------------------------------------------------------------
\begin{frame}{Carbon costs enter once}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\color{policy}\Large\bfseries Policy representation}\par\medskip
      \begin{itemize}\setlength\itemsep{6pt}
        \item CLEWS: an emissions penalty changes technology costs and the optimal build.
        \item OG-Core: a household wedge can represent direct carbon cost and revenue.
        \item Industrial carbon remains in CLEWS while OG production excludes energy.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{claret}\Large\bfseries Accounting condition}\par\medskip
      If the CLEWS reform price includes the penalty, the same resource cost must not
      re-enter OG-Core through a second undifferentiated wedge.
      \medskip

      \begin{alertblock}{Application status}
        The application emits a carbon-price artifact for a future re-solve. Source-reform
        provenance must be confirmed before attributing the macro result to that policy.
      \end{alertblock}
    \end{column}
  \end{columns}
  \source{ogclews_link/channels.py: carbon_tax and emit_carbon_penalty; ogclews_link/experiments.py: coupled; docs/design/methods-paper-review-follow-up.md.}
\end{frame}

\begin{frame}{Capital demand follows the firm's FOC}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      For Cobb--Douglas production, the energy industry's capital demand satisfies
      \[
        K_E = \frac{\gamma_E p_E Y_E}{\rho_E},
        \qquad
        \widehat K_E = \widehat\gamma_E\,\widehat p_E\,\widehat Y_E
      \]
      when the user-cost terms are unchanged between the two solves.
      \smallskip

      A higher capital share need not raise the capital stock. Price and output determine
      sectoral revenue and therefore capital demand.
    \end{column}
    \begin{column}{0.44\textwidth}
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{@{}lr@{}}
        \toprule
        Earlier M=8 solve & Change \\
        \midrule
        capital exponent $\gamma_E$ & $+12.2\%$ \\
        electricity price $p_E$ & $-42.8\%$ \\
        electricity output $Y_E$ & $+20.9\%$ \\
        electricity capital $K_E$ & $-22.4\%$ \\
        identity gap & $0.023$ pp \\
        \bottomrule
      \end{tabular}
      \smallskip

      {\color{claret}\bfseries Evidence status}\par
      {\color{sub}Recomputed from earlier solved outputs; a run under the re-blessed
      baseline is queued.}
    \end{column}
  \end{columns}
  \source{paper/sections/05-channels.tex, Table 2; docs/design/methods-paper-review-response.md.}
\end{frame}

\begin{frame}{Health enters through survival and effective labour}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\color{clews}\Large\bfseries Physical and epidemiological path}\par\medskip
      \begin{itemize}\setlength\itemsep{6pt}
        \item CLEWS supplies the reform/base PM$_{2.5}$ emissions path.
        \item GBD supplies the attributable burden and age profile.
        \item An external dose-response scales emissions into a mortality target.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{ogc}\Large\bfseries Economic path}\par\medskip
      \begin{itemize}\setlength\itemsep{6pt}
        \item mortality changes survival and age composition;
        \item morbidity changes effective labour;
        \item both enter the household and fiscal equilibrium.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  {\color{sub}\centering A $2.7\%$ emissions reduction maps to about 95 fewer deaths per year;
  the morbidity productivity elasticity remains an application assumption.\par}
  \source{paper/sections/05-channels.tex, ``Health''; paper/sections/06-calibration.tex; ogclews_link/data/pm25_health.json.}
\end{frame}

% 10 --------------------------------------------------------------------------
\begin{frame}{Returns and activity flow back to CLEWS}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\color{ogc}\Large\bfseries Discount-rate channel}\par\medskip
      The post-reform return is written to the CLEWS planning-rate input.
      \medskip

      \textbf{Still to establish}
      \begin{itemize}\setlength\itemsep{4pt}
        \item social rate versus financing cost
        \item real/nominal and annualisation conventions
        \item public/private risk and tax adjustments
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{clews}\Large\bfseries Demand channel}\par\medskip
      Sectoral output or consumption scales annual energy-service demand.
      \medskip

      \textbf{Current evidence}
      \begin{itemize}\setlength\itemsep{4pt}
        \item the artifact is produced after the OG solve
        \item a controlled $+10\%$ demand patch raises production $+4.3\%$
        \item the outer controller has not passed the emitted artifact end to end
      \end{itemize}
    \end{column}
  \end{columns}
  \source{paper/sections/05-channels.tex, discount-rate and demand channels; paper/sections/04-framework.tex, architecture; docs/design/INTEGRATION-HANDOFF.md.}
\end{frame}
